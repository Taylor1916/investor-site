Deposit Funds

Select a token and send it using the BNB Smart Chain network.

${message ? `

${escapeHtml(message)}

` : ''}
Network

BNB Smart Chain (BEP20)

Wallet address
${DEPOSIT_ADDRESS}
Important: Send only the selected token on BNB Smart Chain. Check the network before transferring. Sending on the wrong network may result in loss of funds.

Your deposit request is pending review until the payment is verified.

`; document.getElementById('cancelDeposit').addEventListener( 'click', () => loadDashboard() ); document.getElementById('copyDepositAddress').addEventListener( 'click', async () => { try { await navigator.clipboard.writeText(DEPOSIT_ADDRESS); document.getElementById('copyDepositAddress').textContent = 'Address copied!'; } catch { renderDepositForm(data, 'Copy failed. Please copy the address manually.'); } } ); document.getElementById('submitDeposit').addEventListener( 'click', async (e) => { const btn = e.currentTarget; const amount = Number( document.getElementById('depositAmount').value ); const coin = document.getElementById('depositCoin').value; const txHash = document.getElementById('depositTxHash').value.trim(); if (!Number.isFinite(amount) || amount <= 0) { return renderDepositForm(data, 'Enter a valid deposit amount.'); } if (txHash && !/^0x[a-fA-F0-9]{64}$/.test(txHash)) { return renderDepositForm(data, 'Enter a valid transaction hash.'); } btn.disabled = true; btn.textContent = 'Submitting...'; try { const res = await fetch(`${API_BASE}/api/requests`, { method: 'POST', headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${getToken()}` }, body: JSON.stringify({ type: 'deposit', amount, note: [ `Token: ${coin}`, 'Network: BNB Smart Chain (BEP20)', `Wallet: ${DEPOSIT_ADDRESS}`, `Transaction hash: ${txHash || 'Not provided'}` ].join(' | ') }) }); const body = await res.json(); if (!res.ok) { throw new Error(body.message || 'Deposit request failed.'); } flash = body.message || 'Deposit request submitted for verification.'; loadDashboard(); } catch (err) { renderDepositForm(data, err.message); } } ); }
